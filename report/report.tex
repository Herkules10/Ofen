    \section{Introduction}

    % Slide 3
    \begin{frame}
        \frametitle{Introduction - Problem Description}
        \begin{itemize}
            \item \textbf{Given} an image-based dataset with $n$ data points $d_1, d_2, ..., d_n \in \mathbb{R}^{H \times W \times C}$, where $H, W, C$ represent height, width and channels respectively, and $n \in \mathbb{N}$ represents the number of samples.
               \item \textbf{Find} $\alpha = {(\alpha_1, \alpha_2, ..., \alpha_n)}^\top \in \mathcal{D}^n$ which maximizes
                \begin{align*}
                    f\left(\alpha, \mathcal{D}\right) := \lambda_2 \cdot \text{Accuracy}(\alpha, \mathcal{D}) - \lambda_1 \cdot \text{NetworkSize}(\alpha)
                \end{align*} \\
                Where: \\ 
                \begin{itemize}
                    \item $\alpha$ represents the network architecture parameters
                    \item $\mathcal{D}$ is the dataset
                    \item $\lambda_1, \lambda_2$ are weighting factors balancing size vs. accuracy
                    \item NetworkSize$(\alpha)$ counts the total number of parameters
                    \item Accuracy$(\alpha, \mathcal{D})$ is the classification accuracy on dataset $\mathcal{D}$
                \end{itemize}   
        \end{itemize}
    \end{frame}


    % Slide 3a
    \begin{frame}
        \frametitle{Introduction - Motivation}
        \begin{itemize}
            \item For $L$ maximum layers and $C$ component types, exhaustive search requires $\mathcal{O}(C^L)$ evaluations
            \item \textbf{Consequence:} Analytical methods become computationally unviable
            \item \textbf{Solution:} Metaheuristic algorithms (GA, PSO, SA) provide more efficient exploration \cite{idrissi_genetic_2016,kuo_neural_2022,jamous_neural_2021,carvalho_particle_2007,shrestha_optimizing_2019}
            \item \textbf{Expansion:} Consider more layers such as Conv1D/Conv2D, that were not considered in the related work
        \end{itemize}
    \end{frame}

    % Slide 3b
    \begin{frame}
        \frametitle{Introduction - Architecture Components}
        Unlike traditional approaches that only focus on fully connected layers, we extend the search space to include the following layers:
        \begin{itemize}
            \item \textbf{Convolutional Layers (Conv1D/Conv2D):} Feature extraction with learnable filters
            \item \textbf{Fully Connected Layers (FC):} Dense layers for classification
            \item \textbf{Batch Normalization:} Normalization of the values for training stability
            \item \textbf{Max Pooling:} Spatial dimension reduction
            \item \textbf{Dropout:} Regularization technique to avoid overfitting
            \item \textbf{Activation Functions:} ReLU, Tanh, Sigmoid
        \end{itemize}
    \end{frame}

    % Slide 4
    \begin{frame}
        \frametitle{Introduction - Solution Space Characteristics}
        \begin{itemize}
            \item Non-linear cost function $f\left(\alpha, \mathcal{D}\right) := \lambda_2 \cdot \text{Accuracy}(\alpha, \mathcal{D}) - \lambda_1 \cdot \text{NetworkSize}(\alpha)$ combines discrete network size with continuous accuracy metrics
            \item Mixed discrete-continuous solution space consisting of architecture topology (discrete) and hyperparameters (continuous)
            \item Solution space can be represented by encoding each architecture $\alpha$ as a variable-length chromosome
            \item \textbf{Naive Approach:} Exhaustive search with runtime $\mathcal{O}(C^L)$ for $L$ max layers with $C$ component types becomes intractable for realtistic network sizes
        \end{itemize}
    \end{frame}

    % Slide 5 
    \begin{frame}
        \frametitle{Introduction - Project Objectives}
        \begin{itemize}
            \item Investigate how \textbf{Genetic Algorithms (GA)}, \textbf{Particle Swarm Optimization (PSO)}, and \textbf{Simulated Annealing (SA)} perform when applied to the problem.
            \item Balance the trade-off between network size and accuracy
            \item Evaluate the effectiveness of different layer types in automated architecture design
            \item Assess the performance of these algorithms by comparing them to baseline methods, such as \textbf{Random Search (RS)} and \textbf{Grid Search (GS)}.
            \item Assess the performance of the algorithms across different dataset complexities
        \end{itemize}
    \end{frame}

    \section{Algorithms}

    % Slide 6
    \begin{frame}
        \frametitle{Algorithms - Baseline Methods}
        Given a image-based dataset $\mathcal{D} = \{d_1, ..., d_n\}$ \\
        \vspace{2mm}
        \textbf{Random Search (RS)} \\
        {\small
        \textbf{Cost function:} $f(\alpha^{(i)}) = \lambda_2 \cdot \text{Accuracy}(\alpha^{(i)}, \mathcal{D}) - \lambda_1 \cdot \text{NetworkSize}(\alpha^{(i)})$ 
        \begin{enumerate}
            \item Generate $N$ random candidate architectures $\alpha^{(1)}, ..., \alpha^{(N)}$
            \item Train each network and evaluate $f(\alpha^{(i)})$ for $i = 1, 2, ..., N$
            \item Return $(\alpha^{(best, RS)}, f(\alpha^{(best, RS)}))$ with $f(\alpha^{(best, RS)}) \geq f(\alpha^{(i)}), \forall \alpha^{(i)}$
        \end{enumerate}}
        \vspace{2mm}
        \textbf{Grid Search (GS)} \\
        {\small
            \textbf{Cost function:} $f(\alpha^{(i)}) = \lambda_2 \cdot \text{Accuracy}(\alpha^{(i)}, \mathcal{D}) - \lambda_1 \cdot \text{NetworkSize}(\alpha^{(i)})$ 
            \begin{enumerate}
                \item Define systematic grid over architecture hyperparameters 
                \item Evaluate $f(\alpha^{(i)})$ for each grid point $\alpha^{(i)}$
                \item Return $(\alpha^{(best, GS)}, f(\alpha^{(best, GS)}))$ with maximal fitness
            \end{enumerate}}
    \end{frame}

    % Slide 7
    \begin{frame}
        \frametitle{Algorithms - Genetic Algorithm}
        \textbf{Genetic Algorithm (GA) \cite{idrissi_genetic_2016}}: \\
        {
            \footnotesize
            \textbf{Fitness function:} $fitness = \lambda_2 \cdot \text{Accuracy}(\alpha^{(i)}, \mathcal{D}) - \lambda_1 \cdot \text{NetworkSize}(\alpha^{(i)})$
            
            \textbf{Encoding:} Variable-length chromosomes representing layer sequences
            \begin{itemize}
                \item Gene type: \{Conv2D, Conv1D, FC, BatchNorm, MaxPool, Dropout, Activation\}
                \item Gene parameters: kernel\_size, stride, filters, units, etc.
            \end{itemize}

            \textbf{Algorithm Steps:}
            \begin{enumerate}
                \item Create \texttt{population\_size} random architectures $\alpha^{(i)} = (layer_1, layer_2, ..., layer_k)$
                \item Train networks and compute fitness scores
                \item Tournament selection with \texttt{tournament\_size} individuals
                \item Single-point crossover with probability \texttt{crossover\_prob}, preserving valid layer sequences 
                \item Add/remove layers with probability \texttt{mutation\_add\_prob} / \texttt{mutation\_del\_prob} and modify layer parameters with probability \texttt{mutation\_param\_prob}
                \item Generate new population, repeat for \texttt{num\_generations} generations
                \item Return $(\alpha^{(best, GA)}, fitness^{(best, GA)})$
            \end{enumerate}
        }
    \end{frame}

    % Slide 9
    \begin{frame}
        \frametitle{Algorithms - Particle Swarm Optimization}
        \textbf{Particle Swarm Optimization (PSO) \cite{carvalho_particle_2007, jamous_neural_2021}:} \\ 
        {
            \footnotesize
            \textbf{Position Representation:} Continuous vector $x^{(i)} \in \mathbb{R}^d$ encoding the architecture parameters \\

            \textbf{Velocity Update:} $v^{(i)}(t+1) = w \cdot v^{(i)}(t) + c_1 \cdot r_1 \cdot \left(pbest^{(i)} - x^{(i)}(t)\right) + c_2 \cdot r_2 \cdot \left(gbest - x^{(i)}(t)\right)$ \\

            \textbf{Position Update:} $x^{(i)}(t+1) = x^{(i)}(t) + v^{(i)}(t+1)$ \\ 

            \textbf{Architecture Decoding:} Map continuous positions to discrete architectures: 
            \begin{itemize}
                \item Layer count: $\lfloor x_1 \cdot \text{\texttt{max\_layers}} \rfloor$
                \item Layer types: $\text{argmax}\left(\text{softmax}\left([x_2, x_3, ..., x_7]\right)\right)$
                \item Layer parameters: sigmoid/tanh transformations to valid ranges
            \end{itemize} 

            \textbf{Algorithm Steps:}
            \begin{enumerate}
                \item Initialize swarm: Set positions $x^{(i)}(0)$ at random and set velocities $v^{(i)}(0) = 0$
                \item Convert positions to architectures and train the networks 
                \item Update personal best $pbest^{(i)}$ and global best $gbest$ 
                \item Update velocities and positions 
                \item Repeat for \texttt{num\_iterations} iterations 
                \item Return $(\alpha^{(best, PSO)}, fitness^{(best, PSO)})$ 
            \end{enumerate}

        }
    \end{frame}
    
    % Slide 10
    \begin{frame}
        \frametitle{Algorithms - Simulated Annealing}
        \textbf{Simulated Annealing (SA) \cite{kuo_neural_2022}:} \\
        {
            \footnotesize
            \textbf{State Representation:} Current architecture $\alpha_{current}$ \\ 

            \textbf{Neighbourhood Function:} Small architectural modifications: 
            \begin{itemize}
                \item Add/remove single layer 
                \item Modify layer parameters 
                \item Swap adjacent layers
            \end{itemize}

            \textbf{Algorithm Steps:}
            \begin{enumerate}
                \item Start with random architecture $\alpha_\text{current}$
                \item Apply random modification to generate $\alpha_\text{neighbour}$ 
                \item Acceptance Criterion: 
                \begin{itemize}
                    \item If $f(\alpha_\text{neighbour}) < f(\alpha_\text{current})$: accept 
                    \item Else: Accept with probability $e^{-\left(\frac{f(\alpha_\text{neighbour}) - f(\alpha_\text{current})}{T}\right)}$
                \end{itemize}
                \item Reduce temperature $T = T \cdot \text{\texttt{cooling\_rate}}$ every 10 iterations
                \item Repeat until convergence or \texttt{max\_iterations}
                \item Return $(\alpha^{(best, SA)}, fitness^{(best, SA)})$
            \end{enumerate}
        }
    \end{frame}

        \begin{itemize}

                \item \texttt{epochs} = 3 (training) and 20 (evaluation)
                \item \texttt{learning\_rate} = 0.001

            \item Algorithm-specific fixed parameters used in this work:
            \begin{itemize}
                \item GA: \texttt{pop\_size} = 40, \texttt{n\_gen} = 30
                \item PSO: \texttt{swarm\_size} = 40, \texttt{num\_iter} = 30
                \item SA: \texttt{max\_iter} = 1200
            \end{itemize}
        \end{itemize}

    \begin{frame}
        \frametitle{Parametric Optimization - Optimization Methodology}
        \textbf{Simple algorithm to find the best parameters:}
        {
            \small
            \begin{algorithmic}
            \For{each algorithm $\in$ \{GA, PSO, SA\}}
                \For{each parameter\_set $\in$ parameter\_combinations}
                    \For{$k = 1$ to $3$}
                        \State Split dataset into train/validation
                        \State Run algorithm with parameter\_set
                        \State Collect: best\_fitness, convergence\_time, final\_accuracy
                    \EndFor
                    \State Compute: avg\_fitness, avg\_time, avg\_accuracy
                \EndFor
                \State Select parameter\_set with best avg\_fitness
            \EndFor
        \end{algorithmic}
        }

        \textbf{Dataset (CIFAR10 / MNIST) Splits:}
        \begin{itemize}
            \item Training: 60\% (Architecture search)
            \item Validation: 20\% (Fitness evaluation)
            \item Test: 20\% (Final performance assessment)
        \end{itemize}
    \end{frame}

    \section{Conclusion and Discussion}

    \begin{frame}
        \frametitle{Discussion}
        \begin{itemize}
            \item \textbf{Genetic Algorithm (GA):}
            \begin{itemize}
                \item Crossover seems to have a negative influence
                \item More chance to add layers has a positive effect. 
                \item This two things are to be expected, since larger networks should perform better
            \end{itemize}
            \item \textbf{Particle Swarm Optimization (PSO):}
            \begin{itemize}
                \item PSO consistently performed the best for this task across most experiments and is not very volatile to parameter changes.
            \end{itemize}
            \item \textbf{Simulated Annealing (SA):}
            \begin{itemize}
                \item SA performed the worst and can also not be parallelized like the other algorithms.
                \item We would not recommend using SA to optimize neural network architectures.
            \end{itemize}
            \item Experiments were overall success and improvement over simpler methods such as Random Search and Grid Search
            \item CIFAR10 results were rather disappointing, but the MNIST results were good
        \end{itemize}
    \end{frame}

    \begin{frame}
        \frametitle{Conclusion}
        \begin{itemize}
            \small
            \item The architecture of neural networks for a specific dataset was successfully optimized using a genetic algorithm, particle swarm optimization and simulated annealing
            \item A hyperparameter search was conducted and we conclude that the optimization of neural networks seems quite robust to hyperparameter changes in the optimization algorithms
            \item Particle swarm optimization and genetic algorithms present promising pathways to optimize neural network architecture, while simulated annealing seems less promising, especially for larger experiments, as it can't be parallelized
            \item While these results were promising, the immense compute cost severely limits the types of applications these optimization methods could be applied to given current compute limitations
            \item We think the optimization of neural network architecture makes the most sense for somewhat simple problems that require both accuracy and a short inference time, such as collision avoidance for drones
        \end{itemize}
    \end{frame}

    
    \section{Future Work}

    \begin{frame}
        \frametitle{Future Work and Limitations}
        \textbf{Current Limitations:}
        \begin{itemize}
            \item Limited to relatively simple architectures ($\leq$ 10 layers)
            \item Training epochs are reduced for computational feasibility
            \item Single-objective scalarization of multi-objective problem
            \item Runtime of certain approaches became infeasible very quickly
        \end{itemize}
        \vspace{2mm}
        \textbf{Future Extensions:}
        \begin{itemize}
            \item Generalize the approach to different datasets (graph, audio, video, etc)
            \item Introduce more layers to be considered when constructing the NNs
            \item Consider advanced components, such as attention, transformers, etc.
            \item Leverage pre-trained components in search
        \end{itemize}
    \end{frame}
